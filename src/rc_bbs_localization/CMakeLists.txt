cmake_minimum_required(VERSION 3.16)
project(rc_bbs_localization)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
if(NOT CMAKE_BUILD_TYPE)
  set(CMAKE_BUILD_TYPE Release CACHE STRING "Build type" FORCE)
endif()
find_package(ament_cmake REQUIRED)
find_package(rclcpp REQUIRED)
find_package(sensor_msgs REQUIRED)
find_package(nav_msgs REQUIRED)
find_package(geometry_msgs REQUIRED)
find_package(diagnostic_msgs REQUIRED)
find_package(visualization_msgs REQUIRED)
find_package(std_srvs REQUIRED)
find_package(tf2_ros REQUIRED)
find_package(tf2_msgs REQUIRED)
find_package(pcl_conversions REQUIRED)
find_package(PCL REQUIRED COMPONENTS common io filters registration kdtree)
find_package(Eigen3 REQUIRED)
find_package(OpenMP REQUIRED)
find_package(OpenSSL REQUIRED)
find_package(yaml-cpp REQUIRED)

# The pinned upstream files remain pristine. Apply the documented extension in build/.
set(BBS_COPY "${CMAKE_CURRENT_BINARY_DIR}/patched_bbs")
file(REMOVE_RECURSE "${BBS_COPY}")
file(MAKE_DIRECTORY "${BBS_COPY}")
file(COPY third_party/3d_bbs/bbs3d DESTINATION "${BBS_COPY}")
execute_process(COMMAND patch -p1 --batch -i "${CMAKE_CURRENT_SOURCE_DIR}/third_party/cpu_safety.patch"
  WORKING_DIRECTORY "${BBS_COPY}" RESULT_VARIABLE PATCH_RESULT)
if(NOT PATCH_RESULT EQUAL 0)
  message(FATAL_ERROR "Cannot apply pinned BBS CPU extension")
endif()
add_library(rc_official_bbs STATIC
  "${BBS_COPY}/bbs3d/src/cpu_bbs3d/bbs3d.cpp"
  "${BBS_COPY}/bbs3d/src/cpu_bbs3d/voxelmaps.cpp")
target_include_directories(rc_official_bbs PUBLIC "${BBS_COPY}/bbs3d/include" ${PCL_INCLUDE_DIRS})
target_link_libraries(rc_official_bbs PUBLIC OpenMP::OpenMP_CXX Eigen3::Eigen)

add_library(rc_bbs_core STATIC src/backend.cpp src/preprocess.cpp src/engine.cpp src/quality.cpp src/gicp.cpp)
target_include_directories(rc_bbs_core PUBLIC include ${PCL_INCLUDE_DIRS})
target_link_libraries(rc_bbs_core PUBLIC rc_official_bbs ${PCL_LIBRARIES} OpenSSL::Crypto)
target_compile_options(rc_bbs_core PRIVATE -Wall -Wextra)

if(EXISTS "${CMAKE_CURRENT_SOURCE_DIR}/src/node.cpp")
  add_executable(bbs_localizer src/node.cpp)
  target_link_libraries(bbs_localizer rc_bbs_core)
  ament_target_dependencies(bbs_localizer rclcpp sensor_msgs nav_msgs geometry_msgs diagnostic_msgs
    visualization_msgs std_srvs tf2_ros tf2_msgs pcl_conversions)
  install(TARGETS bbs_localizer DESTINATION lib/${PROJECT_NAME})
endif()
if(EXISTS "${CMAKE_CURRENT_SOURCE_DIR}/src/benchmark.cpp")
  add_executable(bbs_benchmark src/benchmark.cpp)
  target_link_libraries(bbs_benchmark rc_bbs_core yaml-cpp)
  install(TARGETS bbs_benchmark DESTINATION lib/${PROJECT_NAME})
endif()
install(FILES LICENSE DESTINATION share/${PROJECT_NAME})
install(DIRECTORY third_party DESTINATION share/${PROJECT_NAME})
install(DIRECTORY scripts/ DESTINATION lib/${PROJECT_NAME} USE_SOURCE_PERMISSIONS)
if(BUILD_TESTING)
  find_package(ament_cmake_gtest REQUIRED)
  if(EXISTS "${CMAKE_CURRENT_SOURCE_DIR}/test/test_core.cpp")
    ament_add_gtest(test_bbs_core test/test_core.cpp)
    target_link_libraries(test_bbs_core rc_bbs_core)
  endif()
endif()
ament_package()
