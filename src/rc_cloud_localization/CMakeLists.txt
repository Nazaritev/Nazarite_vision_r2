cmake_minimum_required(VERSION 3.16)
project(rc_cloud_localization)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
find_package(ament_cmake REQUIRED)
find_package(rclcpp REQUIRED)
find_package(sensor_msgs REQUIRED)
find_package(nav_msgs REQUIRED)
find_package(geometry_msgs REQUIRED)
find_package(diagnostic_msgs REQUIRED)
find_package(visualization_msgs REQUIRED)
find_package(std_srvs REQUIRED)
find_package(tf2_msgs REQUIRED)
find_package(pcl_conversions REQUIRED)
find_package(PCL REQUIRED COMPONENTS common io filters registration kdtree)

add_library(cloud_registration STATIC src/registration.cpp)
target_include_directories(cloud_registration PUBLIC include ${PCL_INCLUDE_DIRS})
target_link_libraries(cloud_registration PUBLIC ${PCL_LIBRARIES})
add_executable(cloud_localizer src/cloud_localizer.cpp)
target_link_libraries(cloud_localizer cloud_registration)
ament_target_dependencies(cloud_localizer rclcpp sensor_msgs nav_msgs geometry_msgs
  diagnostic_msgs visualization_msgs std_srvs tf2_msgs pcl_conversions)
target_compile_options(cloud_localizer PRIVATE -Wall -Wextra -Wpedantic)
install(TARGETS cloud_localizer DESTINATION lib/${PROJECT_NAME})
install(PROGRAMS scripts/cloud_localization_preflight.py DESTINATION lib/${PROJECT_NAME})
if(BUILD_TESTING)
  find_package(ament_cmake_gtest REQUIRED)
  find_package(ament_cmake_pytest REQUIRED)
  ament_add_gtest(test_registration test/test_registration.cpp)
  target_link_libraries(test_registration cloud_registration)
  ament_add_pytest_test(test_preflight_policy test/test_preflight_policy.py)
endif()
ament_package()
